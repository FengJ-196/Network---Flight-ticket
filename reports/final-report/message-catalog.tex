% Slide 6: How to read a message contract
\begin{frame}{How to read a message contract}
\footnotesize\renewcommand{\arraystretch}{1.15}
\begin{tabularx}{\textwidth}{|p{2.5cm}|Y|}
\hline
\labelrow{Request} & One UTF-8 JSON object per line (\msg{\n}) over TCP. Required envelope fields: \msg{type} (uppercase command), \msg{id} (unique per request), \msg{sessionId} (token, or \msg{null} before login), \msg{date} (ISO-8601 send timestamp), and \msg{data} (object). \\
\hline
\labelrow{Reply} & Appends \msg{_RESP} to \msg{type} and echoes \msg{id}, \msg{sessionId}, and \msg{date}. Success sets \msg{code} = \msg{OK}. Failure sets \msg{code} = \msg{E_...}, adds human-readable \msg{message}, and puts error context in \msg{data}. \\
\hline
\labelrow{Push} & Server-initiated \msg{NOTIFY} sets \msg{id} = \msg{null}, includes \msg{sessionId}, \msg{date}, and event payload in \msg{data}, and omits \msg{code}. Clients match replies by \msg{id} and dispatch pushes by \msg{type}; pushes receive no reply. \\
\hline
\labelrow{Field notation} & Contract tables list the fields inside \msg{data}. Fields ending in \msg{?} are optional or conditional; an empty payload is written \texttt{\{\}}. \\
\hline
\end{tabularx}
\end{frame}
