\begin{frame}{Booking lifecycle and system responsibilities}
\small\contractrows
\begin{tabularx}{\textwidth}{|p{2.65cm}|Y|}
\hline
\labelrow{Customer} & Register and log in, search and compare flights, and book tickets using account balance. Can view, print, cancel, or change bookings. \\
\hline
\labelrow{Server} & Listens over TCP, logs all requests and replies, manages mocked flight data, deducts balance on booking, and sends e-tickets via email. \\
\hline
\labelrow{Administrator} & Updates flight status (delay, reschedule, or cancel). The server stores notifications and pushes them to online customers. \\
\hline
\end{tabularx}
\end{frame}

\begin{frame}{How to read a message contract}
\small\contractrows
\begin{tabularx}{\textwidth}{|p{2.65cm}|Y|}
\hline
\labelrow{Request} & One UTF-8 JSON object per line over TCP. Required envelope fields are \msg{type} (uppercase command), \msg{id} (unique on this connection), \msg{sessionId} (session token, \msg{null} before sign-in), \msg{date} (ISO-8601 timestamp when sent) and \msg{data} (object). \\
\hline
\labelrow{Reply} & Append \msg{_RESP} to the command and echo \msg{id}, \msg{sessionId} and \msg{date}. Success sets \msg{code} = OK. Failure sets \msg{code} to an error code (\msg{E_...}), with a human-readable \msg{message} and context in \msg{data}. \\
\hline
\labelrow{Push} & \msg{NOTIFY} uses a null ID. A push may arrive before a pending reply. Match replies by ID and dispatch pushes by type. \\
\hline
\labelrow{Field notation} & The following contracts list the fields inside \msg{data}. Fields marked ? are optional or conditional. An empty object is written \{\}. \\
\hline
\end{tabularx}

\end{frame}

\begin{frame}{Appendix: shared data types}
\small\contractrows
\begin{tabularx}{\textwidth}{|p{2.65cm}|Y|}
\hline
\labelrow{Dates and money} & Dates use \msg{YYYY-MM-DD}; timestamps use ISO-8601 (\msg{+07:00}). All monetary amounts are integer VND (no \msg{currency} field). \\
\hline
\labelrow{Seat and leg} & \msg{seatClass} is \msg{ECONOMY}, \msg{PREMIUM_ECONOMY}, or \msg{BUSINESS}. \msg{legType} is \msg{OUTBOUND} or \msg{RETURN}. \\
\hline
\labelrow{Passenger} & Object with \msg{fullName}, \msg{dob} (\msg{YYYY-MM-DD}), and \msg{idNumber}. \\
\hline
\labelrow{Flight / Offer} & \msg{flightId}, \msg{flightNo}, \msg{airline}, \msg{origin}, \msg{destination}, \msg{departAt}, \msg{arriveAt}, \msg{seatClass}, \msg{price}, \msg{seatsLeft}, \msg{status}. \\
\hline
\end{tabularx}
\end{frame}

\begin{frame}{Appendix: error codes (1/2)}
\footnotesize\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\textwidth}{|l|Y|Y|}
\hline
\labelrow{Code} & \labelrow{When returned} & \labelrow{Client action} \\
\hline
\msg{E_BAD_JSON} & Malformed JSON or invalid UTF-8 frame. & Fix message format and resend. \\
\hline
\msg{E_UNKNOWN_TYPE} & Unrecognized \msg{type} command. & Use a valid command name. \\
\hline
\msg{E_VALIDATION} & Missing or invalid request field. & Prompt user to fix the indicated \msg{field}. \\
\hline
\msg{E_NOT_AUTHENTICATED} & Protected command without valid \msg{sessionId}. & Log in first and retry. \\
\hline
\msg{E_FORBIDDEN} & Signed-in role lacks permission (e.g. non-admin). & Show permission denied message. \\
\hline
\end{tabularx}
\end{frame}

\begin{frame}{Appendix: error codes (2/2)}
\footnotesize\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\textwidth}{|l|Y|Y|}
\hline
\labelrow{Code} & \labelrow{When returned} & \labelrow{Client action} \\
\hline
\msg{E_BAD_CREDENTIALS} & Username or password mismatch on \msg{LOGIN}. & Prompt for credentials again. \\
\hline
\msg{E_USERNAME_TAKEN} & Username already exists on \msg{REGISTER}. & Choose another username. \\
\hline
\msg{E_NOT_FOUND} & Flight or booking does not exist (or not owned). & Refresh list and retry. \\
\hline
\msg{E_SEATS_UNAVAILABLE} & Not enough seats in the requested class. & Choose another flight or class. \\
\hline
\msg{E_INSUFFICIENT_BALANCE} & Account balance is lower than required fare. & Select a cheaper flight or top up. \\
\hline
\msg{E_INTERNAL} & Unexpected server error. & Show error and retry later. \\
\hline
\end{tabularx}
